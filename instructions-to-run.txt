# 4. Test a saved checkpoint (replace 25000 with the step number you want)

# --- Saving to Hugging Face instead of keeping checkpoints locally ---

# 5. Log in to Hugging Face once on the training machine (needs a token with write access)
uv run huggingface-cli login

# 6. Train and upload every checkpoint (every 1000 steps) and the TensorBoard logs to a private HF repo.
#    Only the latest checkpoint stays on local disk (for --resume).
XLA_PYTHON_CLIENT_MEM_FRACTION=0.9 uv run scripts/train.py pi05_g1_dex3_statenoise --exp-name=stage1_noise003 --overwrite --log-interval=50 --hub-repo-id=<your-hf-user>/pi05-g1-dex3-stage1-noise003

# 7. Download one checkpoint from HF to test it (replace the step number)
uv run huggingface-cli download <your-hf-user>/pi05-g1-dex3-stage1-noise003 --include "checkpoints/5000/*" --local-dir hub_checkpoints/stage1_noise003
uv run scripts/test_g1_dex3_checkpoint.py --config-name pi05_g1_dex3_statenoise --checkpoint-dir hub_checkpoints/stage1_noise003/checkpoints/5000

# 8. TensorBoard from HF: download the logs, then open them
uv run huggingface-cli download <your-hf-user>/pi05-g1-dex3-stage1-noise003 --include "runs/*" --local-dir hub_checkpoints/stage1_noise003
uvx tensorboard --logdir hub_checkpoints/stage1_noise003/runs --port 6006
